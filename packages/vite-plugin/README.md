# @devup-api/vite-plugin

devup API plugin for Vite

## Installation

```bash
npm install @devup-api/vite-plugin
```

## Usage

```typescript
import devupApiVitePlugin from '@devup-api/vite-plugin';
import { defineConfig } from 'vite';

export default defineConfig({
  plugins: [
    devupApiVitePlugin({
      // options
    }),
  ],
});
```
